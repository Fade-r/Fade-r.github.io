    def main(cfg: RadarConfig) -> None:
    output_dirs = ensure_output_dirs(cfg)
    raw = load_raw_int16(cfg.bin_path)
    validation_report = validate_layout_candidates(raw, cfg)
    adc_cube, parsing_summary = parse_adc_matlab(raw, cfg)

    if cfg.frame_idx >= adc_cube.shape[0]:
        raise IndexError(
            f"frame_idx={cfg.frame_idx} is out of range"
        )

    frame0 = frame_to_matlab_view(adc_cube, 0)
    frame1 = frame_to_matlab_view(adc_cube, 1 if adc_cube.shape[0] > 1 else 0)
    selected_frame = frame_to_matlab_view(adc_cube, cfg.frame_idx)
    range_profile = run_range_fft(selected_frame, cfg)
    speed_profile = run_doppler_fft_strict(range_profile, cfg)
    angle_profile = run_angle_fft_strict(speed_profile, cfg)
    angle_heatmap = compute_range_angle_heatmap(angle_profile)

    summaries = summarize_range_over_frames(adc_cube, cfg)
    average_range_profile = summaries["average_range_profile"]
    range_time_map = summaries["range_time_map"]
    channel_traces = summaries["channel_traces"]
    slow_time_trace = extract_slow_time(channel_traces, cfg)
    time_axis = build_capture_time_axis(cfg, adc_cube.shape[0])
    phase, displacement = compute_phase_displacement(slow_time_trace, cfg)
    doppler = compute_doppler_spectrum(slow_time_trace, cfg, time_axis)
    spectrogram = compute_spectrogram(slow_time_trace, cfg, time_axis)
    channel_correlation = compute_channel_correlation(channel_traces)

    读这段代码时要看见的两个分支¶

    selected_frame
  └─ range FFT
       ├─ strict Doppler → strict angle → range-angle diagnostic
       └─ 100-frame summaries
            ├─ average range / range-time
            ├─ selected-bin phase / displacement
            ├─ interpolated spectrum / STFT
            └─ channel normalized inner product

    validation_report 与实际解析之间没有分支连接，这是当前最重要的控制流缺口。

    %% 多普勒FFT
speed_profile = zeros(parameter.Samples,parameter.Chirps,parameter.txNum*parameter.rxNum);
for k=1:parameter.rxNum
    for n=1:parameter.rangeBin
      temp=range_profile(n,:,k).*(doppler_win)';
      temp_fft=fftshift(fft(temp,parameter.dopplerBin));
      speed_profile(n,:,k)=temp_fft;
    end
end

%% 方位角FFT
angle_profile = zeros(parameter.Samples,parameter.Chirps,parameter.angleBin);
for n=1:parameter.rangeBin
    for m=1:parameter.dopplerBin
      temp=speed_profile(n,m,:);
      temp=temp(1:8);
      temp_fft=fftshift(fft(temp,parameter.angleBin));
      angle_profile(n,m,:)=temp_fft;
    end
end

    输出数据契约¶

    selected_frame_debug.npz 包含 17 个数组，核心 shape 为：

    key shape 说明
    selected_frame (256,128,12) strict 单帧 cube
    range_profile (256,128,12) Range FFT
    speed_profile (256,128,12) strict Doppler；后 8 通道为零
    angle_profile (256,128,180) 8→180 空间 FFT
    range_time_map (100,256) frame × range
    channel_traces (12,12800) 固定 range bin 慢时间
    doppler_spectrum (49628,) 跨 gap 插值后频谱
    spectrogram_matrix (256,772) 插值序列 STFT

    NPZ 约 96.3 MiB，不应直接载入网页。Wiki 只使用预生成 PNG、shape 和经过审计的标量。

    下一版控制流应怎样写¶

    report = validate_layout_candidates(raw, cfg)

if cfg.processing_mode == "strict_matlab":
    adc_cube = parse_adc_matlab(raw, cfg)
elif cfg.processing_mode == "tdm_candidate":
    require_calibration_passed(cfg)
    adc_cube = parse_adc_tdm(raw, cfg)
else:
    raise ValueError("Explicit processing_mode required")

    关键不是自动选择最高分，而是让 layout 成为 显式、可追踪、只有在标定通过后才能启用 的实验配置。
